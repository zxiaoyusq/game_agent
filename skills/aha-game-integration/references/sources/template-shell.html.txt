      window.h5sdk.gameLoadingCompleted();

      }

    function getParamsByUrl() {
      const search = '?' + location.search.slice(1).replace(/\?/g, '&')
      const params = {}
      if (search) {
        search
          .substr(1)
          .split('&')
          .filter((item) => item)
          .forEach((item) => {
            const [key, value] = item.split('=')
            if (params[key]) {
              const temp = params[key].split(',')
              temp.push(value)
              params[key] = temp.join(',')
            } else {
              params[key] = value
            }
          })
      }
      return params
    }
